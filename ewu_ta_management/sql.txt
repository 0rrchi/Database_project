DROP DATABASE IF EXISTS ewu_ta_management;

CREATE DATABASE ewu_ta_management;

USE ewu_ta_management;


-- =====================================================
-- DEPARTMENT
-- =====================================================

CREATE TABLE Department
(
    DepartmentID INT AUTO_INCREMENT PRIMARY KEY,

    DepartmentName VARCHAR(100) NOT NULL UNIQUE,

    FloorNo VARCHAR(100)
);


-- =====================================================
-- ADMIN
-- =====================================================

CREATE TABLE Admin
(
    AdminID VARCHAR(20) PRIMARY KEY,

    Password VARCHAR(255) NOT NULL,

    FirstName VARCHAR(50) NOT NULL,

    LastName VARCHAR(50) NOT NULL,

    Email VARCHAR(100) NOT NULL UNIQUE
);


-- =====================================================
-- STUDENT
-- =====================================================

CREATE TABLE Student
(
    StudentID VARCHAR(20) PRIMARY KEY,

    FirstName VARCHAR(50) NOT NULL,

    LastName VARCHAR(50) NOT NULL,

    Email VARCHAR(100) NOT NULL UNIQUE,

    Phone VARCHAR(20),

    DateOfBirth DATE NOT NULL,

    CGPA DECIMAL(3,2) NOT NULL,

    Password VARCHAR(255),

    IsActivated BOOLEAN DEFAULT FALSE,

    DepartmentID INT NOT NULL,

    FOREIGN KEY (DepartmentID)
    REFERENCES Department(DepartmentID)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);


-- =====================================================
-- FACULTY
-- =====================================================

CREATE TABLE Faculty
(
    FacultyID VARCHAR(20) PRIMARY KEY,

    FirstName VARCHAR(50) NOT NULL,

    LastName VARCHAR(50) NOT NULL,

    Email VARCHAR(100) NOT NULL UNIQUE,

    Designation VARCHAR(50),

    Password VARCHAR(255),

    IsActivated BOOLEAN DEFAULT FALSE,

    RecruitingTA BOOLEAN NOT NULL DEFAULT FALSE,

    DepartmentID INT NOT NULL,

    FOREIGN KEY (DepartmentID)
    REFERENCES Department(DepartmentID)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);


-- =====================================================
-- COURSE
-- =====================================================

CREATE TABLE Course
(
    CourseID INT AUTO_INCREMENT PRIMARY KEY,

    CourseCode VARCHAR(20) NOT NULL UNIQUE,

    CourseTitle VARCHAR(100) NOT NULL,

    Credit DECIMAL(2,1) NOT NULL,

    DepartmentID INT NOT NULL,

    FOREIGN KEY (DepartmentID)
    REFERENCES Department(DepartmentID)
    ON DELETE CASCADE
    ON UPDATE CASCADE
);


-- =====================================================
-- FACULTY COURSE
--
-- Stores which courses a faculty is teaching
-- in a particular semester.
-- =====================================================

CREATE TABLE Faculty_Course
(
    FacultyCourseID INT AUTO_INCREMENT PRIMARY KEY,

    FacultyID VARCHAR(20) NOT NULL,

    CourseID INT NOT NULL,

   
    FOREIGN KEY (FacultyID)
    REFERENCES Faculty(FacultyID)
    ON DELETE CASCADE
    ON UPDATE CASCADE,

    FOREIGN KEY (CourseID)
    REFERENCES Course(CourseID)
    ON DELETE CASCADE
    ON UPDATE CASCADE,

    CONSTRAINT unique_faculty_course
    UNIQUE
    (
    
       FacultyID ,

    CourseID 
    )
);


-- =====================================================
-- STUDENT COURSE
--
-- Stores courses completed/taken by students.
-- This is used to determine whether a student is
-- eligible to apply to a faculty.
-- =====================================================

CREATE TABLE Student_Course
(
    StudentCourseID INT AUTO_INCREMENT PRIMARY KEY,

    StudentID VARCHAR(20) NOT NULL,

    CourseID INT NOT NULL,

    SemesterName ENUM
    (
        'Spring',
        'Summer',
        'Fall'
    ) NOT NULL,

    Year YEAR NOT NULL,

    Grade VARCHAR(5),

    FOREIGN KEY (StudentID)
    REFERENCES Student(StudentID)
    ON DELETE CASCADE
    ON UPDATE CASCADE,

    FOREIGN KEY (CourseID)
    REFERENCES Course(CourseID)
    ON DELETE CASCADE
    ON UPDATE CASCADE,

    CONSTRAINT unique_student_course
    UNIQUE
    (
        StudentID,
        CourseID,
        SemesterName,
        Year
    )
);


-- =====================================================
-- TA APPLICATION
--
-- IMPORTANT:
-- Student applies directly to a FACULTY.
--
-- There is NO FacultyCourseID here.
-- There is NO CourseID here.
-- =====================================================

CREATE TABLE TA_Application
(
    ApplicationID INT AUTO_INCREMENT PRIMARY KEY,

    StudentID VARCHAR(20) NOT NULL,

    FacultyID VARCHAR(20) NOT NULL,

    ResumeFile VARCHAR(255),

    CoverLetter VARCHAR(255),

    ApplicationDate DATE NOT NULL,

    Status ENUM
    (
        'Pending',
        'Accepted',
        'Rejected'
    ) DEFAULT 'Pending',

    FOREIGN KEY (StudentID)
    REFERENCES Student(StudentID)
    ON DELETE CASCADE
    ON UPDATE CASCADE,

    FOREIGN KEY (FacultyID)
    REFERENCES Faculty(FacultyID)
    ON DELETE CASCADE
    ON UPDATE CASCADE,

    CONSTRAINT unique_student_faculty_application
    UNIQUE
    (
        StudentID,
        FacultyID
    )
);
